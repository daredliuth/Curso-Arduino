\section{Clase 1: Fundamentos de programación}
\subsection{Variables}
\begin{frame}[fragile]{Variables}
    En C, una variable es un espacio de memoria en el cuál se guardan datos para ser utilizados después. \cite{gfg_variables}
    \\
    Las variables se definen mediante la siguiente sintáxix:
    \begin{verbatim}
        tipo nombre = valor;
    \end{verbatim}
    \begin{exampleblock}{Ejemplos}
        \begin{verbatim}
        int edad = 45;
        float decimales = 1.15;
        char letra = 'a';
        bool verdad = true;
        \end{verbatim}
    \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Variables}
    Reglas para nombrar variables:
    \begin{itemize}
        \item Solo pueden contener letras, dígitos y guión bajo.
        \item No pueden comenzar con un dígito.
        \item No pueden ser \textbf{palabras reservadas}.
        \item El nombre debe ser único en el programa.
    \end{itemize}
\end{frame}

\subsection{Asignación y expresiones}
\begin{frame}[fragile]{Asignación}
    Es una operación en la cual el valor del operando derecho
    se asigna al operando izquierdo (variable).\cite{microsoft_asignacion}
    \begin{exampleblock}{Ejemplos}
        \begin{verbatim}
                =, +=, -=
                *=, /=, %=
            <<=, >>=, &=, ^=, |=
        \end{verbatim}
    \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Expresiones}
    Es una secuencia de operadores y sus operandos, que especifica un cómputo.\cite{cpp_expressions}
    Se pueden clasificar como unarias, binarias, y otras.
    \begin{block}{Unarias}
        \begin{verbatim}
                ++. --, !
        \end{verbatim}
    \end{block}
    \begin{block}{Binarias}
        \begin{verbatim}
        +. -. ==, !=, <, > <=, >=
        \end{verbatim}
    \end{block}
\end{frame}
